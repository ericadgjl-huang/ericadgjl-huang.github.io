EPPS 6356 Assignment 6: Shiny Apps

Overview

In this assignment I moved from running sample Shiny apps to publishing my own. I changed the fonts of a sample app in two ways, built three apps on R’s built-in datasets, built one app on my own data, and published two of them on shinyapps.io. The two published apps are embedded on this page, so they can be used directly here.

All my apps share one style file, eric_theme.R. It uses the same warm colours as my Assignment 5 charts (burnt orange #C8551E, warm dark grey text and light gridlines), the Inter font from Google Fonts, and a ggplot2 theme with no box and a left-aligned title. Each app folder keeps its own copy of this file, so that every app can be deployed on its own.

Show eric_theme.R 
# =============================================================
# eric_theme.R — my Shiny + ggplot2 style for EPPS 6356
# Same colours as eric_style.R (Assignment 5): clean-modern layout,
# burnt-orange accent, light warm-grey gridlines, no box.
# Each app folder keeps its own copy so the app can be deployed alone.
# =============================================================

eric_col <- list(
  accent = "#C8551E",  # burnt orange: single-series marks
  tint   = "#ECC4B0",  # light version of the accent: background groups
  ink    = "#2B2622",  # titles, axis titles
  ink2   = "#5E5650",  # tick labels, legend text
  muted  = "#8C837C",  # source notes
  grid   = "#ECE6E0",  # gridlines
  base   = "#CFC6BE",  # axis lines
  bg     = "#FFFFFF"
)

# Categorical colours: use in order. Colour 3 (gold) is low-contrast on white.
eric_cat <- c("#C8551E", "#00897B", "#C68F00", "#3A6FB0",
              "#D45F86", "#5E8A1A", "#7B4FB0", "#A8432B")

# ---- App theme (bslib) -----------------------------------------
eric_bs_theme <- function(...) {
  bslib::bs_theme(
    version      = 5,
    bg           = eric_col$bg,
    fg           = eric_col$ink,
    primary      = eric_col$accent,
    base_font    = bslib::font_google("Inter", wght = c(400, 600, 700)),
    heading_font = bslib::font_google("Inter", wght = c(400, 600, 700)),
    "headings-font-weight" = 700,
    ...
  )
}

# thematic needs ragg or showtext to draw Google fonts in plots. This block
# never runs; it only tells rsconnect to install them on shinyapps.io.
if (FALSE) {
  library(ragg)
  library(showtext)
}

# Plots follow the app theme (font = "auto" also passes Inter to the plots)
eric_thematic <- function() {
  thematic::thematic_shiny(bg = eric_col$bg, fg = eric_col$ink,
                           accent = eric_col$accent, font = "auto")
}

# ---- ggplot2 theme ---------------------------------------------
# Only theme() tweaks, no complete theme such as theme_minimal(), so the
# font that thematic sets is kept.
theme_mine <- function(grid = "y") {
  grid_line <- ggplot2::element_line(colour = eric_col$grid, linewidth = 0.5)
  ggplot2::theme(
    plot.background    = ggplot2::element_rect(fill = eric_col$bg, colour = NA),
    panel.background   = ggplot2::element_rect(fill = eric_col$bg, colour = NA),
    panel.border       = ggplot2::element_blank(),
    panel.grid.minor   = ggplot2::element_blank(),
    panel.grid.major.x = if (grepl("x", grid)) grid_line else ggplot2::element_blank(),
    panel.grid.major.y = if (grepl("y", grid)) grid_line else ggplot2::element_blank(),
    axis.line.x        = ggplot2::element_line(colour = eric_col$base, linewidth = 0.4),
    axis.ticks         = ggplot2::element_line(colour = eric_col$base, linewidth = 0.4),
    axis.ticks.length  = ggplot2::unit(2.5, "pt"),
    axis.text          = ggplot2::element_text(colour = eric_col$ink2, size = ggplot2::rel(0.9)),
    axis.title         = ggplot2::element_text(colour = eric_col$ink),
    plot.title         = ggplot2::element_text(colour = eric_col$ink, face = "bold",
                                               size = ggplot2::rel(1.2), hjust = 0,
                                               margin = ggplot2::margin(b = 4)),
    plot.subtitle      = ggplot2::element_text(colour = eric_col$ink2, hjust = 0,
                                               margin = ggplot2::margin(b = 8)),
    plot.caption       = ggplot2::element_text(colour = eric_col$muted, size = ggplot2::rel(0.75),
                                               hjust = 0, margin = ggplot2::margin(t = 10)),
    plot.title.position   = "plot",
    plot.caption.position = "plot",
    legend.position    = "top",
    legend.justification = "left",
    legend.key         = ggplot2::element_blank(),
    legend.background  = ggplot2::element_blank(),
    legend.title       = ggplot2::element_text(colour = eric_col$ink2, size = ggplot2::rel(0.9)),
    legend.text        = ggplot2::element_text(colour = eric_col$ink2, size = ggplot2::rel(0.9)),
    plot.margin        = ggplot2::margin(10, 16, 8, 10)
  )
}

Part 1. Sample apps and my own fonts

I ran the sample apps from the Teams folder (01_hello to 05_sliders). Then I changed 04_mpg, which compares fuel economy across groups of cars, to use my own fonts in two ways.

i. CSS. I added a tags$style() block to the UI. The body text uses Segoe UI and the headings use Georgia. Each rule lists several fonts and ends with the generic family (sans-serif or serif), because a visitor’s computer may not have my first choice.

    paste("mpg ~", input$variable)
  })

  output$caption <- renderText({
    formulaText()
  })

  output$mpgPlot <- renderPlot({
    boxplot(as.formula(formulaText()),
            data = mpgData,
            outline = input$outliers,
            col = "#75AADB", pch = 19)
  })
}

ii. bslib. I passed a bs_theme() with Inter from font_google() to the page, and added thematic::thematic_shiny(font = "auto") to the server. The font = "auto" argument is needed: without it, thematic changes the plot colours but not the plot font. I also restyled the boxplot with my own colours, gridlines, title and source note.

library(datasets)
library(bslib)

# ---- My colours (same values as eric_style.R) ----------------------------
eric_col <- list(

  titlePanel("Miles Per Gallon (bslib fonts)"),

  sidebarLayout(
    sidebarPanel(
      selectInput("variable", "Variable:",
                  c("Cylinders" = "cyl",
                    "Transmission" = "am",
                    "Gears" = "gear")),
      checkboxInput("outliers", "Show outliers", TRUE)
    ),
    mainPanel(
      h3(textOutput("caption")),
      plotOutput("mpgPlot")
    )

server <- function(input, output) {

  # ---- Make the plots follow the app theme -------------------------------
  # font = "auto" is needed: by default thematic changes the plot colours
  # but leaves the plot font alone.
  thematic::thematic_shiny(
    bg = eric_col$bg, fg = eric_col$ink, accent = eric_col$accent,
    font = "auto"
  )
  # ------------------------------------------------------------------------

  formulaText <- reactive({
    paste("mpg ~", input$variable)
  })

  output$caption <- renderText({
    formulaText()
  })

  output$mpgPlot <- renderPlot({
    f <- as.formula(formulaText())

    # Layout only (colours and font come from thematic)
    op <- par(las = 1, mgp = c(2.6, 0.55, 0), tcl = -0.25,
              mar = c(5.5, 4.5, 3.5, 1.5))
    on.exit(par(op))

    # 1) invisible pass to set up the axes, 2) gridlines, 3) the boxes on top
    boxplot(f, data = mpgData, outline = input$outliers,
            col = NA, border = NA, outpch = NA,
            axes = FALSE, frame = FALSE, xlab = "", ylab = "")
    abline(h = axTicks(2), col = eric_col$grid, lwd = 1)
    boxplot(f, data = mpgData, outline = input$outliers, add = TRUE,
            col = eric_col$accent, border = eric_col$ink2,
            medcol = "white", medlwd = 2.5,
            whisklty = 1, staplewex = 0.4, pch = 19, outcol = eric_col$ink2,
            axes = TRUE, frame = FALSE)
    # boxplot(add = TRUE) does not draw axis labels, so add them here
    title(xlab = x_labels[[input$variable]],
          ylab = "Fuel economy (miles per US gallon)",
          col.lab = eric_col$ink)

    # Left-aligned title and source note, as in my Assignment 5 charts
    x0 <- grconvertX(0.02, from = "nfc", to = "user")
    mtext("Fuel economy by car characteristic, 1973–74 models",
          side = 3, at = x0, adj = 0, line = 1.4, font = 2, cex = 1.1,
          col = eric_col$ink)
    mtext("Source: Motor Trend (1974), via the mtcars dataset in R",
          side = 1, at = x0, adj = 0, line = 4.2, cex = 0.72,
          col = eric_col$muted)
  }, res = 96)
}

When I ran both versions, the difference was clear. With CSS, the title and the controls changed, but the text inside the plot (the axis labels and numbers) stayed in R’s default font. With bslib and thematic, the page and the plot used the same font.

c. Which method would I use? For a real project I would use bslib. A single bs_theme() call sets the fonts and colours for the whole app, and thematic_shiny() passes them to the plots, so the text inside the charts matches the page. CSS cannot do this, because a Shiny plot is an image drawn by R and CSS only styles the web page around it. In addition, font_google() delivers the font with the app, so every visitor sees the same typeface.

Part 2. Three apps on built-in datasets

I gave each app a different layout and different kinds of inputs, so that they are not the same app with the data swapped. Every app has at least two inputs, a plot plus a table or text summary, and one reactive() that feeds several outputs. Every chart has a title, axis labels with units and a source note.

a. mtcars: what goes with fuel economy?

This app plots fuel economy against a variable the user chooses (weight, horsepower, engine size, quarter-mile time or rear axle ratio). The points can be coloured by cylinders or transmission, and a slider keeps only cars in a horsepower range. The same filtered data feed the plot, the correlation sentence and the table, so the three always agree. I chose a scatter plot on purpose, so that the app is clearly different from the boxplot in the 04_mpg sample, which uses the same data.

Show the mtcars app 
# =============================================================
# Assignment 6 · Part 2a — mtcars: what goes with fuel economy?
# Layout: sidebar (bslib::page_sidebar)
# Inputs: x variable, colour grouping, horsepower range, trend line
# Outputs: scatter plot, text summary, table — all from one reactive()
# =============================================================

library(shiny)
library(bslib)
library(ggplot2)
source("eric_theme.R")

# ---- Data ----------------------------------------------------
cars <- mtcars
cars$car <- rownames(mtcars)
cars$cyl_f <- factor(cars$cyl, labels = c("4 cylinders", "6 cylinders", "8 cylinders"))
cars$am_f  <- factor(cars$am,  labels = c("Automatic", "Manual"))

# x-axis choices: column = c(menu label, axis label with units)
x_vars <- list(
  wt   = c("Weight",            "Weight (1,000 lbs)"),
  hp   = c("Horsepower",        "Gross horsepower (hp)"),
  disp = c("Engine size",       "Engine displacement (cubic inches)"),
  qsec = c("Quarter-mile time", "Quarter-mile time (seconds)"),
  drat = c("Rear axle ratio",   "Rear axle ratio (ratio)")
)
x_choices <- setNames(names(x_vars), sapply(x_vars, `[`, 1))

source_note <- "Source: Motor Trend road tests of 1973–74 models, via R's mtcars dataset"

# ---- UI ------------------------------------------------------
ui <- page_sidebar(
  title = "What goes with fuel economy? 32 cars from 1973–74",
  theme = eric_bs_theme(),
  fillable = FALSE,   # let cards keep their natural height

  sidebar = sidebar(
    width = 280,
    selectInput("xvar", "Compare fuel economy with:", choices = x_choices,
                selected = "wt"),
    radioButtons("colour_by", "Colour points by:",
                 choices = c("Cylinders" = "cyl_f",
                             "Transmission" = "am_f",
                             "No grouping" = "none")),
    sliderInput("hp_range", "Keep cars with horsepower between:",
                min = 50, max = 340, value = c(50, 340), step = 10,
                post = " hp"),
    checkboxInput("show_fit", "Show linear trend line", value = TRUE)
  ),

  card(
    card_header("Fuel economy against the selected variable"),
    plotOutput("scatter", height = "420px"),
    textOutput("summary")
  ),
  card(
    card_header("Cars in the current selection"),
    card_body(fillable = FALSE, DT::DTOutput("car_table"))
  )
)

# ---- Server --------------------------------------------------
server <- function(input, output, session) {

  eric_thematic()

  # One reactive() feeds all three outputs: the plot, the summary and the table
  selected_cars <- reactive({
    d <- cars[cars$hp >= input$hp_range[1] & cars$hp <= input$hp_range[2], ]
    validate(need(nrow(d) >= 3,
                  "Fewer than 3 cars in this horsepower range. Widen the slider."))
    d
  })

  x_label <- reactive(x_vars[[input$xvar]][2])
  x_name  <- reactive(tolower(x_vars[[input$xvar]][1]))

  output$scatter <- renderPlot({
    d <- selected_cars()
    d$x <- d[[input$xvar]]

    p <- ggplot(d, aes(x = x, y = mpg))

    if (input$show_fit) {
      p <- p + geom_smooth(method = "lm", formula = y ~ x, se = TRUE,
                           colour = eric_col$ink2, fill = eric_col$grid,
                           linewidth = 0.7)
    }

    if (input$colour_by == "none") {
      p <- p + geom_point(shape = 21, size = 3.2, stroke = 0.3,
                          fill = eric_col$accent, colour = eric_col$ink2)
    } else {
      p <- p +
        geom_point(aes(fill = .data[[input$colour_by]]),
                   shape = 21, size = 3.2, stroke = 0.3, colour = eric_col$ink2) +
        scale_fill_manual(values = eric_cat, name = NULL)
    }

    p +
      labs(title    = paste0("Fuel economy vs ", x_name()),
           subtitle = paste0("Each point is one car model (", nrow(d), " of 32 shown)"),
           x = x_label(),
           y = "Fuel economy (miles per US gallon)",
           caption = source_note) +
      theme_mine(grid = "xy")
  }, res = 96)

  output$summary <- renderText({
    d <- selected_cars()
    r <- cor(d[[input$xvar]], d$mpg)
    direction <- if (r < 0) "lower" else "higher"
    sprintf("%d of 32 cars are shown. Correlation between %s and fuel economy: r = %.2f, so cars with a larger %s tend to have %s fuel economy.",
            nrow(d), x_name(), r, x_name(), direction)
  })

  output$car_table <- DT::renderDT({
    d <- selected_cars()
    out <- data.frame(
      Car          = d$car,
      MPG          = d$mpg,
      Selected     = d[[input$xvar]],
      Cylinders    = d$cyl,
      Transmission = as.character(d$am_f),
      Horsepower   = d$hp
    )
    names(out)[3] <- x_label()
    out <- out[order(-out$MPG), ]
    DT::datatable(out, rownames = FALSE, fillContainer = FALSE,
                  options = list(pageLength = 8, dom = "ftip"))
  })
}

shinyApp(ui, server)

b. USArrests: which states had the most arrests in 1973?

This app ranks the 50 states by murder, assault or rape arrests per 100,000 residents. The user chooses how many states to highlight and whether to show the highest or the lowest rates. The three tabs show a ranked bar chart, a scatter plot against urban population, and the full data table. The same states are highlighted on every tab, because all three read from one reactive().

Show the USArrests app 
# =============================================================
# Assignment 6 · Part 2b — USArrests: which states had the most arrests?
# Layout: navigation bar with tabs (bslib::page_navbar)
# Inputs: crime type, number of states, sort order
# Outputs: ranked bar chart, scatter plot, summary text, table —
#          all from one reactive()
# =============================================================

library(shiny)
library(bslib)
library(ggplot2)
library(ggrepel)   # labels that do not overlap
source("eric_theme.R")

# ---- Data ----------------------------------------------------
arrests <- USArrests
arrests$State <- rownames(USArrests)

crime_labels <- c(Murder  = "Murder",
                  Assault = "Assault",
                  Rape    = "Rape")

source_note <- paste0("Source: World Almanac and Book of Facts 1975 (arrests); ",
                      "Statistical Abstract of the United States 1975\n",
                      "(urban population); via R's USArrests dataset")

# ---- UI ------------------------------------------------------
ui <- page_navbar(
  title = "US arrests by state, 1973",
  theme = eric_bs_theme(),
  fillable = FALSE,   # let cards keep their natural height

  # The sidebar is shared by all tabs, so every tab uses the same selection
  sidebar = sidebar(
    width = 270,
    radioButtons("crime", "Type of arrest:", choices = crime_labels),
    sliderInput("top_n", "Number of states to highlight:",
                min = 5, max = 50, value = 10, step = 1),
    radioButtons("order", "Show states with:",
                 choices = c("The highest rates" = "high",
                             "The lowest rates"  = "low")),
    hr(),
    p(class = "small text-muted",
      "Rates are arrests per 100,000 residents. The same states are",
      "highlighted on every tab.")
  ),

  nav_panel(
    "Ranking",
    card(
      plotOutput("bar_chart", height = "auto"),
      textOutput("summary")
    )
  ),
  nav_panel(
    "Arrests vs urban population",
    card(
      plotOutput("scatter", height = "480px"),
      textOutput("scatter_note")
    )
  ),
  nav_panel(
    "Data",
    card(card_body(fillable = FALSE, DT::DTOutput("state_table")))
  )
)

# ---- Server --------------------------------------------------
server <- function(input, output, session) {

  eric_thematic()

  # One reactive() with the ranking; every output reads from it
  ranked <- reactive({
    d <- arrests
    d$Rate <- d[[input$crime]]
    d <- d[order(d$Rate, decreasing = (input$order == "high")), ]
    d$Rank <- seq_len(nrow(d))
    d$Highlighted <- d$Rank <= input$top_n
    d
  })

  crime_name <- reactive(crime_labels[[input$crime]])
  which_end  <- reactive(if (input$order == "high") "highest" else "lowest")

  output$bar_chart <- renderPlot({
    d <- ranked()
    d <- d[d$Highlighted, ]
    # Highest rate at the top of the chart when sorting high-to-low
    d$State <- factor(d$State, levels = rev(d$State))

    ggplot(d, aes(x = Rate, y = State)) +
      geom_col(fill = eric_col$accent, width = 0.7) +
      geom_text(aes(label = formatC(Rate, format = "f",
                                    digits = if (input$crime == "Assault") 0 else 1)),
                hjust = -0.2, size = 3.3, colour = eric_col$ink2) +
      scale_x_continuous(expand = expansion(mult = c(0, 0.12))) +
      labs(title    = if (nrow(d) == 50)
                        sprintf("All 50 states ranked by %s arrest rate", tolower(crime_name()))
                      else sprintf("The %d states with the %s %s arrest rates",
                                   nrow(d), which_end(), tolower(crime_name())),
           subtitle = "Arrests per 100,000 residents, 1973",
           x = sprintf("%s arrests (per 100,000 residents)", crime_name()),
           y = NULL,
           caption = source_note) +
      theme_mine(grid = "x") +
      theme(axis.line.x = element_blank(),
            axis.ticks.y = element_blank(),
            axis.text.y = element_text(colour = eric_col$ink))
  },
  # Taller chart when more states are shown
  height = function() 150 + 22 * input$top_n,
  res = 96)

  output$summary <- renderText({
    d <- ranked()
    sprintf("Average %s arrest rate: %.1f per 100,000 across all 50 states, and %.1f across the %d states shown.",
            tolower(crime_name()), mean(d$Rate),
            mean(d$Rate[d$Highlighted]), sum(d$Highlighted))
  })

  output$scatter <- renderPlot({
    d <- ranked()
    hl <- d[d$Highlighted, ]

    ggplot(d, aes(x = UrbanPop, y = Rate)) +
      geom_point(data = d[!d$Highlighted, ], colour = eric_col$tint, size = 2.6) +
      geom_point(data = hl, colour = eric_col$accent, size = 2.8) +
      geom_text_repel(data = hl, aes(label = State), size = 3,
                      colour = eric_col$ink2, min.segment.length = 0.2,
                      segment.colour = eric_col$base, seed = 1) +
      labs(title    = sprintf("%s arrests and urbanisation", crime_name()),
           subtitle = sprintf("Orange: the %d states with the %s rates", nrow(hl), which_end()),
           x = "Urban population (% of state residents)",
           y = sprintf("%s arrests (per 100,000 residents)", crime_name()),
           caption = source_note) +
      theme_mine(grid = "xy")
  }, res = 96)

One problem appeared during deployment. rsconnect only installs packages that are named in the code, but thematic needs ragg or showtext to draw Google fonts inside plots. Without them, the published plots would fall back to the default font. I solved this by adding library(ragg) and library(showtext) inside an if (FALSE) block in eric_theme.R: the block never runs, but rsconnect sees the two packages and installs them on the server.

My own app uses the Happy Planet Index 2026 data file (154 countries, 2006–2025), the same source as Assignments 2 and 5, but now as a full panel over time. A short script, prep_data.R, turns the “All Data” sheet of the Excel file into a tidy CSV, so the app does not need the Excel file or readxl.

  • Three or more inputs, one not a drop-down: none of them is a drop-down. There is a year slider (with a play button), radio buttons for the x-axis variable, checkboxes for the regions, and a click on the scatter plot.

# ---- Server --------------------------------------------------
server <- function(input, output, session) {

  eric_thematic()

  # Shared reactive: one year, selected regions. Feeds the scatter plot,
  # the summary, the rankings table and the download.
  year_data <- reactive({
    validate(need(length(input$regions) > 0, "Select at least one region."))
    d <- hpi[hpi$year == input$year & hpi$region %in% input$regions, ]
    d$region <- factor(d$region, levels = regions)
    d
  })

  # Data actually plotted (drops missing values of the chosen x variable)
  plot_data <- reactive({
    d <- year_data()
    d$x <- d[[input$xvar]]
    d[!is.na(d$x), ]
  })

  # Country picked by clicking the scatter plot
  selected_country <- reactiveVal("Costa Rica")
  observeEvent(input$scatter_click, {
    hit <- nearPoints(plot_data(), input$scatter_click,
                      xvar = "x", yvar = "hpi", threshold = 12, maxpoints = 1)
    if (nrow(hit) == 1) selected_country(hit$country)
  })

  x_label <- reactive(x_vars[[input$xvar]][2])
  x_name  <- reactive({
    n <- x_vars[[input$xvar]][1]
    if (input$xvar == "gdp_pc") n else tolower(n)
  })

  # ---- Explore tab ----
  output$scatter <- renderPlot({
    d   <- plot_data()
    sel <- d[d$country == selected_country(), ]

    p <- ggplot(d, aes(x = x, y = hpi)) +
      geom_point(aes(size = population, fill = region),
                 shape = 21, colour = "white", stroke = 0.3, alpha = 0.9) +
      # Ring around the clicked country (fixed size so small countries show too)
      geom_point(data = sel, shape = 21, size = 6, fill = NA,
                 colour = eric_col$ink, stroke = 1.2) +
      geom_text(data = sel, aes(label = country), vjust = -1.4,
                size = 3.6, fontface = "bold", colour = eric_col$ink) +
      scale_fill_manual(values = region_cols, name = NULL, drop = TRUE) +
      scale_size(range = c(1.5, 14), guide = "none") +
      scale_y_continuous(expand = expansion(mult = c(0.05, 0.12))) +
      guides(fill = guide_legend(override.aes = list(size = 3.5), ncol = 2)) +
      coord_cartesian(clip = "off") +   # country labels near the edge stay visible
      labs(title    = sprintf("Happy Planet Index vs %s, %d", x_name(), input$year),
           subtitle = sprintf("%d countries; point size shows population", nrow(d)),
           x = x_label(), y = y_label, caption = source_note) +
      theme_mine(grid = "xy")

    if (input$xvar == "gdp_pc") {
      p <- p + scale_x_log10(labels = function(v) paste0("$", format(v, big.mark = ",",
                                                                    scientific = FALSE, trim = TRUE)))
    }
    p
  }, res = 96)

  output$summary <- renderText({
    d <- plot_data()
    top <- d[which.max(d$hpi), ]
    sprintf("%d countries shown for %d. Highest HPI: %s (%.1f). Correlation between %s and HPI: r = %.2f.",
            nrow(d), input$year, top$country, top$hpi, x_name(),
            cor(if (input$xvar == "gdp_pc") log10(d$x) else d$x, d$hpi))
  })

  output$trend <- renderPlot({
    ctry <- selected_country()
    one  <- hpi[hpi$country == ctry, ]
    reg  <- one$region[1]
    reg_avg <- aggregate(hpi ~ year, data = hpi[hpi$region == reg, ], FUN = mean)

    ggplot() +
      geom_vline(xintercept = input$year, colour = eric_col$base, linetype = "dotted") +
      geom_line(data = reg_avg, aes(x = year, y = hpi),
                colour = eric_col$muted, linewidth = 0.8, linetype = "dashed") +
      geom_line(data = one, aes(x = year, y = hpi),
                colour = eric_col$accent, linewidth = 1.2) +
      geom_point(data = one, aes(x = year, y = hpi),
                 colour = eric_col$accent, size = 1.8) +
      annotate("text", x = max(reg_avg$year), y = tail(reg_avg$hpi, 1),
               label = paste(reg, "average"), hjust = 1, vjust = -0.8,
               size = 3.2, colour = eric_col$muted) +
      scale_x_continuous(breaks = seq(2006, 2025, by = 3), limits = c(2006, 2025)) +
      labs(title    = sprintf("%s: HPI score, 2006–2025", ctry),
           subtitle = sprintf("Dashed: %s average (unweighted). Dotted: %d.",
                              reg, input$year),
           x = "Year", y = "HPI score (0–100)", caption = source_note) +
      theme_mine(grid = "y")
  }, res = 96)

  # ---- Rankings tab ----
  rank_table <- reactive({
    d <- year_data()
    d <- d[order(-d$hpi), ]
    data.frame(
      Rank = seq_len(nrow(d)),
      Country = d$country,
      Region = as.character(d$region),
      `HPI score` = round(d$hpi, 1),
      `Life expectancy (years)` = round(d$life_exp, 1),
      `Life satisfaction (0-10)` = round(d$life_sat, 2),
      `Ecological footprint (gha/person)` = round(d$footprint, 2),
      `GDP per capita (PPP 2021 $)` = round(d$gdp_pc),
      check.names = FALSE
    )
  })

  output$table_title <- renderText(
    sprintf("HPI rankings, %d (%d countries)", input$year, nrow(rank_table()))
  )

  output$rank_table <- DT::renderDT({
    DT::datatable(rank_table(), rownames = FALSE, fillContainer = FALSE,
                  options = list(pageLength = 15, dom = "ftip"))
  })

  output$download <- downloadHandler(
    filename = function() sprintf("hpi_rankings_%d.csv", input$year),
    content  = function(file) write.csv(rank_table(), file, row.names = FALSE)
  )
}
